library(DESeq2)

awk 'BEGIN {FS=OFS="\t"} {zero=1; for (i=2; i<=NF; i++) if ($i != 0) zero=0; if (!zero) print}' RawCountFile_rsemgenes.txt > filtered_counts.txt

count_data <- read.table("filtered_counts.txt", header=TRUE, row.names=1)

control_columns <- 2:4
condition1_columns <- 5:7
condition2_columns <- 8:10

count_data_rounded <- round(count_data)

rs <- SummarizedExperiment(assays = list(counts = as.matrix(count_data_rounded)))

col_data <- data.frame(condition = factor(c(rep("Control", 3), rep("Condition1", 3), rep("Condition2", 3))))

dds <- DESeqDataSetFromMatrix(countData = count_data,
                              colData = col_data,
                              design = ~ condition)

dds <- DESeq(dds)

results_condition1_vs_control <- results(dds, contrast = c("condition", "condition1", "control"))
results_condition2_vs_control <- results(dds, contrast = c("condition", "condition2", "control"))
results_condition1_vs_condition2 <- results(dds, contrast = c("condition", "condition1", "condition2"))

DEGs_condition1_vs_control <- subset(results_condition1_vs_control, padj < 0.05 & abs(log2FoldChange) > 1)
DEGs_condition2_vs_control <- subset(results_condition2_vs_control, padj < 0.05 & abs(log2FoldChange) > 1)
DEGs_condition1_vs_condition2 <- subset(results_condition1_vs_condition2, padj < 0.05 & abs(log2FoldChange) > 1)

results_condition1_vs_control_filtered <- subset(results_condition1_vs_control, !is.na(log2FoldChange) & !is.na(padj) & abs(log2FoldChange) <= 20)

custom_colors <- c(DEGs = "red", Non_DEGs = "black")

DEGs_condition1_vs_control <- subset(results_condition1_vs_control, padj < 0.05 & abs(log2FoldChange) > 1)

write.csv(DEGs_condition1_vs_control, file = "DEGs_condition1_vs_control.csv", row.names = TRUE)

file.exists("DEGs_condition1_vs_control.csv")

volcano_plot_condition1_vs_condition2 <- ggplot(results_condition1_vs_condition2_filtered, aes(x = log2FoldChange, y = -log10(padj), color = ifelse(padj < 0.05 & abs(log2FoldChange) > 1, "DEGs", "Non_DEGs"))) +
  geom_point(size = 1) +
  labs(title = "Volcano Plot - Condition 1 vs. Condition 2",
